
for disk in /dev/sd{a,b,c,d,e,f,g}; do
  parted -s $disk mklabel gpt
  parted -s $disk mkpart primary 0% 100%
  parted -s $disk set 1 raid on
done

for disk in /dev/sd{a,b,c,d,e,f,g}1; do
  wipefs -a $disk
  mdadm --zero-superblock $disk
done

mdadm --create /dev/md0                                    \
  --level=6                                                 \
  --raid-devices=6                                           \
  /dev/sda1 /dev/sdb1 /dev/sdc1 /dev/sdd1 /dev/sde1 /dev/sdf1 \
  --spare-devices=1                                            \
  /dev/sdg1                                                     \
  --metadata=1.2                                                 \
  --assume-clean

mkfs.btrfs -L main /dev/md0

mount /dev/disk/by-label/main /mnt
cd /mnt
btrfs su cr @
btrfs su cr @home

... console=ttyS0,115200

mount -o rw,noatime,compress=zstd:2,ssd,discard=async,space_cache=v2,subvol=@  /dev/disk/by-label/main /mnt
mkdir /mnt/home
mount -o rw,noatime,compress=zstd:2,ssd,discard=async,space_cache=v2,subvol=@home  /dev/disk/by-label/main /mnt/home
mkdir /mnt/boot
mount /dev/sdh1 /mnt/boot
mkdir /mnt/boot/efi
mount /dev/sdh2 /mnt/boot/efi


mount -o rw,noatime,compress=zstd:2,ssd,discard=async,space_cache=v2,subvol=@  /dev/disk/by-label/main /mnt
mount -o rw,noatime,compress=zstd:2,ssd,discard=async,space_cache=v2,subvol=@home  /dev/disk/by-label/main /mnt/home
mount /dev/sdh1 /mnt/boot
mount /dev/sdh2 /mnt/boot/efi
